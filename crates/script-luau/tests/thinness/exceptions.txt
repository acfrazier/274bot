# Rule T6 (ADR 0023 item 20): each Request or Await without a JS v2
# counterpart names the live cell that runs it from Luau.
# the wait for the next game tick queues no request; JS v2 gets the same pause from an awaited Promise (design-r3.md 7.4 rule T6 names it as the one entry at the start).
wait: host.luau-lifecycle
# the typed native plan can select and walk to a world bank; JS v2 open-booth needs a selected booth and has different scope.
bank.openNearest: host.luau-bank-interact
# the typed native plan deposits every bankable backpack item; JS v2 deposit names one item.
bank.depositAll: host.luau-bank-interact
# the host-js type census has no Bank.close declaration; the JavaScript Bank.close shim awaits its bank_close machine, while NativeOp:close is a distinct fire-and-forget InteractReq::Close row.
bank.close: host.luau-bank-interact
# the native binding requires an exact slot and item id; JS v2 held accepts only a name and action.
interact.held: host.luau-interactions
# this generic loc action has no matching JS v2 request; open-booth is bank-only.
interact.loc: host.luau-bank-interact
# JS v2 has no typed NPC action request.
interact.npc: host.luau-interactions
# JS v2 has no typed ground-item request with an exact tile and id.
interact.obj: host.luau-interactions
# JS v2 has no typed player action request.
interact.player: host.luau-interactions
